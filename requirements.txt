# Python 3.10 or 3.11 ONLY. DeepFilterNet has no Windows wheels for 3.12+
# (pip would try to compile it from Rust source and fail).
#
# On Windows/CPU, install torch first from the CPU index (smaller download):
#   pip install torch==2.5.1 torchaudio==2.5.1 --index-url https://download.pytorch.org/whl/cpu
# then: pip install -r requirements.txt

# Deep learning runtime — pinned: pyannote 3.3, SpeechBrain 1.0 and
# DeepFilterNet 0.5.6 are all known to work together on this version.
torch==2.5.1
torchaudio==2.5.1
numpy>=1.24,<2            # DeepFilterNet requires numpy < 2

hf_xet                    # faster model downloads from the HF hub

scipy
soundfile

pyyaml

deepfilternet==0.5.6
noisereduce               # classical spectral-gating baseline

# Stage 2 — diarization

# Stage 3 — speaker embeddings
speechbrain>=1.0,<1.1

# Stage 4 — transcription
faster-whisper>=1.1

# Evaluation
jiwer                     # WER

matplotlib
